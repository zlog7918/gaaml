% superscript cite
\newcommand{\sscite}[2][]{%
  \foreach \citekey in {#2} {%
    \ifx\citekey\empty%
      % do nothing
    \else%
      \textsuperscript{%
        \edef\tmp{\noexpand\cite[#1]{\citekey}}%
        \tmp%
      }%
    \fi%
  }%
}
\pgfkeys{
  /citebook/.is family,
  /citebook,
  author/.estore in=\citebookauthor,
  cite/.estore in=\citebookcite,
  both/.estore in=\citebookboth,
}
\NewDocumentCommand{\citebook}{O{}m}{%
  \def\citebookauthor{}%
  \def\citebookcite{}%
  \def\citebookboth{}%
  \pgfkeys{/citebook,#1}%
  \ifx\citebookboth\empty%
    \ifx\citebookauthor\empty%
      \citeauthor*{#2}%
    \else%
      \citeauthor*[\citebookauthor]{#2}%
    \fi%
    \ifx\citebookcite\empty%
      \sscite{#2}%
    \else%
      \sscite[\citebookcite]{#2}%
    \fi%
  \else%
    \citeauthor*[\citebookboth]{#2}%
    \sscite[\citebookboth]{#2}%
  \fi%
}
